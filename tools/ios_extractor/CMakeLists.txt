# OpenGOAL's level extractor (common + decompiler libraries) as one static library for the iOS app.
# The app calls it from jakIOS/Import/Jak1LevelExtractor.cpp at ISO import.
#
# OpenGOAL's own top-level CMake also configures the game (SDL, curl, ...), which does not build for iOS,
# so this project compiles only what level extraction needs. Source lists are read from OpenGOAL's
# common/ and decompiler/ CMakeLists, so they follow upstream. Build with build.sh (also builds a
# macOS variant for testing).
cmake_minimum_required(VERSION 3.20)
project(jak1_ios_extractor C CXX)

set(OG "${CMAKE_CURRENT_SOURCE_DIR}/../../OpenGOAL")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Reads the source list of add_library(<name> ...) from an OpenGOAL CMakeLists.txt.
function(og_library_sources out cmake_file library)
  file(READ "${cmake_file}" text)
  string(REGEX MATCH "add_library\\([ \t\r\n]*${library}[ \t\r\n]+([^)]*)\\)" match "${text}")
  if(NOT match)
    message(FATAL_ERROR "add_library(${library} ...) not found in ${cmake_file}")
  endif()
  string(REGEX REPLACE "[ \t\r\n]+" ";" items "${CMAKE_MATCH_1}")
  get_filename_component(dir "${cmake_file}" DIRECTORY)
  set(sources "")
  foreach(item IN LISTS items)
    if(item MATCHES "\\.(c|cc|cpp)$")
      list(APPEND sources "${dir}/${item}")
    endif()
  endforeach()
  list(REMOVE_DUPLICATES sources)
  set(${out} "${sources}" PARENT_SCOPE)
endfunction()

og_library_sources(COMMON_SOURCES "${OG}/common/CMakeLists.txt" common)
og_library_sources(DECOMP_SOURCES "${OG}/decompiler/CMakeLists.txt" decomp)

# Not needed for level extraction, and they pull in desktop-only dependencies (REPL, sockets, file
# dialogs, sqlite, terminal control, the game's sound library for audio ripping). ios_stubs.cpp stands in
# for the few functions the remaining sources reference.
list(FILTER COMMON_SOURCES EXCLUDE REGEX "/common/(repl|cross_sockets|sqlite)/")
list(FILTER COMMON_SOURCES EXCLUDE REGEX "/common/util/(dialogs|term_util)\\.cpp$")
list(FILTER DECOMP_SOURCES EXCLUDE REGEX "/decompiler/(decompilation_process|data/streamed_audio)\\.cpp$")

file(GLOB ZSTD_SOURCES "${OG}/third-party/zstd/lib/common/*.c"
                       "${OG}/third-party/zstd/lib/compress/*.c"
                       "${OG}/third-party/zstd/lib/decompress/*.c")

add_library(jak1_extractor STATIC
  ${COMMON_SOURCES}
  ${DECOMP_SOURCES}
  ${ZSTD_SOURCES}
  "${OG}/third-party/fmt/src/format.cc"
  "${OG}/third-party/fmt/src/os.cc"
  "${OG}/third-party/lzokay/lzokay.cpp"
  "${OG}/third-party/stb_image/stb_image.cpp"
  "${OG}/third-party/tiny_gltf/tiny_gltf.cpp"
  "${OG}/third-party/tree-sitter/tree-sitter/lib/src/lib.c"
  "${OG}/third-party/tree-sitter/tree-sitter-opengoal/parser.c"
  ios_stubs.cpp  # REPL and terminal stand-ins
  # xdelta3 is not included: the app already compiles it (jakIOS/Runtime/Jak1XDelta3.c).
)

target_include_directories(jak1_extractor PRIVATE
  "${OG}"
  "${OG}/third-party"
  "${OG}/third-party/fmt/include"
  "${OG}/third-party/zstd/lib"
  "${OG}/third-party/zstd/lib/common"
  "${OG}/third-party/stb_image"
  "${OG}/third-party/tree-sitter/tree-sitter/lib/include"
  "${OG}/third-party/tree-sitter/tree-sitter/lib/src"
)
# common/versions/revision.h is generated by OpenGOAL's configure step (absent in a fresh checkout).
target_include_directories(jak1_extractor BEFORE PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/generated/common/versions/revision.h"
     "#pragma once\n#define BUILT_TAG \"ios\"\n#define BUILT_SHA \"ios\"\n")
target_compile_options(jak1_extractor PRIVATE -w -O2)
# Record source paths (__FILE__ in asserts, debug info) relative to the repo, so the library doesn't embed the
# builder's home directory.
get_filename_component(REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
target_compile_options(jak1_extractor PRIVATE "-ffile-prefix-map=${REPO_ROOT}/=")
